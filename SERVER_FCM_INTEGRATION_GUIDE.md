# OnFood (Buvva) — Python FastAPI Server FCM & Data Messaging Guide

This guide details all necessary changes required in your **Python (FastAPI + SQLAlchemy Async + PostgreSQL)** server (`onfoodserver`) to support **Hybrid FCM and High-Priority Data Messages** for the Android app.

---

## 1. Add Dependencies

Add `firebase-admin` to `requirements.txt`:

```text
firebase-admin>=6.5.0
```

Install via pip:
```bash
pip install firebase-admin
```

---

## 2. Firebase Admin Credentials Setup

1. Go to **Firebase Console $\rightarrow$ Project Settings $\rightarrow$ Service Accounts $\rightarrow$ Generate New Private Key**.
2. Save the downloaded JSON file as `serviceAccountKey.json` inside your server's root or config directory (e.g. `onfoodserver/serviceAccountKey.json`).
3. Add `serviceAccountKey.json` to your `.gitignore`.
4. In `app/config.py`, add:
   ```python
   FIREBASE_CREDENTIALS_PATH: str = os.getenv("FIREBASE_CREDENTIALS_PATH", "serviceAccountKey.json")
   ```

## 3. Database Schema: SQLAlchemy Model

Add the `UserFcmToken` model to `app/models.py`:

```python
# ─────────────────────────────────────────────
# FCM Device Tokens (1 User -> Multiple Devices)
# ─────────────────────────────────────────────

class UserFcmToken(Base):
    __tablename__ = "user_fcm_tokens"

    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    user_id = Column(String, ForeignKey("users.id", ondelete="CASCADE"), nullable=False, index=True)
    fcm_token = Column(Text, nullable=False, unique=True, index=True)
    device_name = Column(String, nullable=True, default="Android Device")
    platform = Column(String, nullable=False, default="android")
    created_at = Column(DateTime, nullable=False, server_default=func.now())
    updated_at = Column(DateTime, nullable=False, server_default=func.now(), onupdate=func.now())

    user = relationship("User", backref="fcm_tokens")
```

### PostgreSQL Migration Script (or Alembic Migration)
Run this SQL script against your PostgreSQL database:

```sql
CREATE TABLE IF NOT EXISTS user_fcm_tokens (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id VARCHAR NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    fcm_token TEXT NOT NULL UNIQUE,
    device_name VARCHAR(100) DEFAULT 'Android Device',
    platform VARCHAR(20) NOT NULL DEFAULT 'android',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX IF NOT EXISTS idx_user_fcm_tokens_user_id ON user_fcm_tokens(user_id);
CREATE INDEX IF NOT EXISTS idx_user_fcm_tokens_token ON user_fcm_tokens(fcm_token);
```

## 5. API Endpoints in `app/routers/auth.py`

Add the registration and unregistration endpoints into `app/routers/auth.py`:

```python
from app.models import UserFcmToken
from app.schemas import DeviceTokenRequest
from sqlalchemy.dialects.postgresql import insert as pg_insert
import firebase_admin
from firebase_admin import firestore

@router.post("/device-token", status_code=status.HTTP_200_OK)
async def register_device_token(
    request: DeviceTokenRequest,
    user_id: str = Depends(get_current_user_id_verified),
    db: AsyncSession = Depends(get_db)
):
    """
    Register or update an FCM device token for the authenticated user.
    1. Stores in PostgreSQL user_fcm_tokens table
    2. Stores in Cloud Firestore under users/{user_id} collection
    """
    token = request.fcmToken.strip()
    if not token:
        raise BadRequestException("fcmToken cannot be empty")

    # 1. Store in PostgreSQL
    stmt = pg_insert(UserFcmToken).values(
        user_id=user_id,
        fcm_token=token,
        device_name=request.deviceName or "Android Device",
        platform=request.platform or "android"
    ).on_conflict_do_update(
        index_elements=["fcm_token"],
        set_={
            "user_id": user_id,
            "device_name": request.deviceName or "Android Device",
            "platform": request.platform or "android",
            "updated_at": datetime.datetime.utcnow()
        }
    )
    await db.execute(stmt)
    await db.commit()

    # 2. Store in Cloud Firestore
    try:
        if firebase_admin._apps:
            fs = firestore.client()
            user_ref = fs.collection("users").document(user_id)
            user_ref.set({
                "userId": user_id,
                "fcmTokens": firestore.ArrayUnion([token]),
                "lastActive": firestore.SERVER_TIMESTAMP,
                "devices": {
                    token.replace(".", "_")[-24:]: {
                        "token": token,
                        "deviceName": request.deviceName or "Android Device",
                        "platform": request.platform or "android",
                        "updatedAt": firestore.SERVER_TIMESTAMP
                    }
                }
            }, merge=True)
    except Exception as e:
        logger.warning(f"[Firestore] Failed to sync token to Firestore: {e}")

    return {"success": True, "message": "FCM device token registered in PostgreSQL and Firestore"}


@router.delete("/device-token", status_code=status.HTTP_200_OK)
async def unregister_device_token(
    request: DeviceTokenRequest,
    user_id: str = Depends(get_current_user_id_verified),
    db: AsyncSession = Depends(get_db)
):
    """
    Remove an FCM device token upon user logout from both PostgreSQL and Firestore.
    """
    token = request.fcmToken.strip()
    if token:
        # 1. Delete from PostgreSQL
        query = select(UserFcmToken).where(
            UserFcmToken.fcm_token == token,
            UserFcmToken.user_id == user_id
        )
        result = await db.execute(query)
        record = result.scalars().first()
        if record:
            await db.delete(record)
            await db.commit()

        # 2. Delete from Cloud Firestore
        try:
            if firebase_admin._apps:
                fs = firestore.client()
                user_ref = fs.collection("users").document(user_id)
                user_ref.update({
                    "fcmTokens": firestore.ArrayRemove([token]),
                    f"devices.{token.replace('.', '_')[-24:]}": firestore.DELETE_FIELD
                })
        except Exception as e:
            logger.warning(f"[Firestore] Failed to remove token from Firestore: {e}")

    return {"success": True, "message": "Device token removed successfully from PostgreSQL and Firestore"}
```

## 6. FCM Service Module (`app/services/fcm.py`)

Create `app/services/fcm.py` to handle Firebase Admin initialization and high-priority multicast dispatch:

```python
import os
import asyncio
import logging
from typing import List, Dict, Any, Optional
import firebase_admin
from firebase_admin import credentials, messaging
from sqlalchemy.future import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.models import UserFcmToken
from app.config import settings

logger = logging.getLogger(__name__)

# Initialize Firebase Admin SDK once on module load
try:
    cred_path = getattr(settings, "FIREBASE_CREDENTIALS_PATH", "serviceAccountKey.json")
    if os.path.exists(cred_path) and not firebase_admin._apps:
        cred = credentials.Certificate(cred_path)
        firebase_admin.initialize_app(cred)
        logger.info("[FCM] Firebase Admin SDK initialized successfully")
    elif not os.path.exists(cred_path):
        logger.warning(f"[FCM] Firebase credentials not found at '{cred_path}'. Push notifications disabled.")
except Exception as e:
    logger.error(f"[FCM] Failed to initialize Firebase Admin SDK: {e}")


def _build_status_copy(status: str, token_display: str) -> tuple[str, str]:
    """Return friendly (title, message) for each status."""
    st = status.upper()
    if st == "PREPARING":
        return "Kitchen is Cooking 🍳", f"Token #{token_display} is being prepared by the chef!"
    elif st in ("READY", "READY_FOR_PICKUP"):
        return "Order Ready to Collect! 🔔", f"Token #{token_display} is ready at the counter!"
    elif st in ("DELIVERED", "CLOSED"):
        return "Order Collected ✅", f"Enjoy your meal! Token #{token_display} completed."
    elif st in ("CANCELLED", "REJECTED", "FAILED"):
        return "Order Cancelled", f"Order #{token_display} has been cancelled."
    return "Order Update", f"Your order status is now: {status}"


async def send_order_status_push_async(
    db: AsyncSession,
    user_id: str,
    order_id: str,
    order_token: str,
    status: str,
    estimated_ready_at: Optional[str] = None,
    items_details: Optional[str] = None
):
    """
    Queries active FCM tokens for the given user_id and dispatches a high-priority
    DATA-ONLY FCM message. Runs on a threadpool to prevent blocking the async loop.
    """
    if not firebase_admin._apps:
        return

    # 1. Fetch user's registered FCM tokens
    stmt = select(UserFcmToken.fcm_token).where(UserFcmToken.user_id == user_id)
    result = await db.execute(stmt)
    tokens: List[str] = result.scalars().all()

    if not tokens:
        logger.debug(f"[FCM] No registered device tokens for user_id={user_id}")
        return

    title, message = _build_status_copy(status, order_token or "")

    # 2. Build High-Priority DATA payload (No top-level 'notification' block!)
    data_payload = {
        "type": "ORDER_STATUS_UPDATE",
        "order_id": str(order_id),
        "order_token": str(order_token or ""),
        "status": str(status).upper(),
        "estimated_ready_at": str(estimated_ready_at or ""),
        "items_details": str(items_details or "Food tray items"),
        "title": title,
        "message": message,
    }

    # 3. Dispatch using send_each_for_multicast via run_in_executor
    loop = asyncio.get_running_loop()
    await loop.run_in_executor(None, _dispatch_multicast_sync, tokens, data_payload)


def _dispatch_multicast_sync(tokens: List[str], data_payload: Dict[str, str]):
    try:
        message = messaging.MulticastMessage(
            tokens=tokens,
            data=data_payload,
            android=messaging.AndroidConfig(
                priority="high",
                ttl=86400  # 24 hours in seconds
            )
        )
        response = messaging.send_each_for_multicast(message)
        logger.info(f"[FCM] Sent {response.success_count}/{len(tokens)} push messages for order {data_payload.get('order_id')}")

        # Check for invalid/stale tokens
        if response.failure_count > 0:
            stale_tokens = []
            for idx, resp in enumerate(response.responses):
                if not resp.success:
                    err_code = getattr(resp.exception, "code", "")
                    if "registration-token-not-registered" in str(err_code) or "invalid-registration-token" in str(err_code):
                        stale_tokens.append(tokens[idx])
            
            if stale_tokens:
                logger.info(f"[FCM] Stale tokens to prune: {len(stale_tokens)}")
                # Pruning in Firestore
                try:
                    if firebase_admin._apps and user_id:
                        fs = firestore.client()
                        fs.collection("users").document(user_id).update({
                            "fcmTokens": firestore.ArrayRemove(stale_tokens)
                        })
                except Exception as e:
                    logger.warning(f"[Firestore] Failed to prune tokens: {e}")
    except Exception as e:
        logger.error(f"[FCM] Failed to dispatch multicast push: {e}")
```

## 7. Cloud Firestore Document Structure

When a user logs in, their document in Cloud Firestore (`users/{userId}`) is automatically structured like this:

```json
users/
  └── "usr_101"/
        ├── userId: "usr_101"
        ├── lastActive: September 28, 2026 at 05:40:00 UTC
        ├── fcmTokens: [
        │     "eX_abc123...456",
        │     "fM_xyz888...999"
        │   ]
        └── devices: {
              "c123...456": {
                "token": "eX_abc123...456",
                "deviceName": "Samsung Galaxy S23",
                "platform": "android",
                "updatedAt": September 28, 2026 at 05:40:00 UTC
              }
            }
```

## 8. Connecting FCM to Your Existing Order Event Bridge (`app/main.py`)

In `onfoodserver/app/main.py`, you already have:

```python
async def handle_incoming_event(event_data):
    if event_data.get("event") == "order_status_updated":
        data = event_data.get("data")
        user_id = data.get("userId")
        if user_id:
            # Existing SSE and WebSocket broadcast
            await sse_manager.broadcast_to_user(user_id, "order-status", data)
            await ws_manager.broadcast_to_user(user_id, data)

            # ──────────────────────────────────────────────────────────
            # NEW: Trigger High-Priority FCM Push for Mobile Devices
            # ──────────────────────────────────────────────────────────
            from app.database import AsyncSessionLocal
            from app.services.fcm import send_order_status_push_async

            async with AsyncSessionLocal() as session:
                order_token = str(data.get("orderToken") or data.get("pickupNumber") or "")
                await send_order_status_push_async(
                    db=session,
                    user_id=user_id,
                    order_id=str(data.get("id") or data.get("orderId")),
                    order_token=order_token,
                    status=str(data.get("status")),
                    estimated_ready_at=data.get("estimatedReadyAt"),
                    items_details=data.get("itemsDetails")
                )
```

With this single integration in `handle_incoming_event`:
- **Every time an order status updates anywhere in the backend** (whether triggered by vendor app, kitchen staff, or scheduled job), the event automatically dispatches to **both SSE (for live in-app watching) and FCM (for phone locked / background delivery)**!
